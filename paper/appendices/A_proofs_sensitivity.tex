\subsection{解的适定性}
\label{sec:proof:model-positive}
\begin{proposition}
\label{prop:model:positive}
设参数与初值满足第~\ref{sec:model:equations}~节的假设，$c(t)\ge0$ 局部有界且可测，$q(t)\in[0,1]$ 可测，则
\begin{enumerate}
\renewcommand{\labelenumi}{(\roman{enumi})}
\item 式\eqref{eq:model:full}在 $[0,\infty)$ 上存在唯一的解，它在每个有界区间上绝对连续，且几乎处处满足方程；
\item 对一切 $t\ge0$，各仓室非负、总和为 $N$，且 $S(t),I(t)>0$。
\end{enumerate}
\end{proposition}

\begin{proof}
由 $c$ 局部有界可测、$q\in[0,1]$ 可测，式\eqref{eq:model:full}的右端关于时间可测。右端各分量是状态的至多二次多项式，系数在有限时间区间上有界，因此在有限时间区间与有界状态集上，右端及其关于状态的 Lipschitz 常数都有界。由 Carathéodory 存在唯一性定理，局部解存在唯一，它绝对连续且几乎处处满足方程。

沿局部解分别积分 $S$ 与 $I$ 的线性方程，得
\begin{align*}
S(t)&=S_0\exp\!\left[-\int_0^t\frac{c(s)[\beta+(1-\beta)q(s)]}{N}I(s)\,ds\right],\\
I(t)&=I_0\exp\!\left[\int_0^t\left(\frac{\beta c(s)(1-q(s))}{N}S(s)-\gamma\right)\,ds\right],
\end{align*}
由 $S_0,I_0>0$，在解的存在区间上 $S,I>0$。$S_q$ 的流入项 $(1-\beta)cqSI/N$ 与初值均非负，故 $S_q\ge0$。$I_q$ 的方程是线性方程，非齐次项 $\beta cqSI/N$ 与初值都非负，由常数变易公式，$I_q\ge0$。于是 $R$ 的流入 $\gamma I+\delta_qI_q$ 非负，$R\ge0$。

五个方程相加，右端为零，故总和恒为 $N$。由非负性与总人口守恒，解始终位于紧集 $[0,N]^5$，故由延拓定理可延拓至 $[0,\infty)$。
\end{proof}

\subsection{触发与退出}
\label{sec:proof:threshold-rationale}
\begin{proof}[\normalfont\bfseries 命题~\ref{prop:model:threshold-rationale}的证明]
(i) 由 $S^*>S_c$ 与式\eqref{eq:model:Sc}，
\[
\dot I(t_1)=\eta\left[\frac{\beta c_0(1-q_0)S^*}{N}-\gamma\right]>0,
\]
故继续采用 $(c_0,q_0)$ 时，存在 $\varepsilon>0$，使 $I$ 在 $(t_1,t_1+\varepsilon)$ 上大于 $\eta$。

(ii) 自 $t_a$ 起实施常规控制，由 $S,I>0$（命题~\ref{prop:model:positive}）得
\[
\dot I=\gamma I\left(\frac{S}{S_c}-1\right),\qquad
\dot S=-\frac{c_0[\beta+(1-\beta)q_0]}{N}SI<0.
\]
若 $S(t_a)\le S_c$，则 $S$ 此后严格下降，$t>t_a$ 时都有 $S<S_c$ 与 $\dot I<0$，$I$ 不再超过 $\eta$。若 $S(t_a)>S_c$，则由连续性，$t_a$ 的某个右邻域内仍有 $S>S_c$，从而 $\dot I>0$，$I$ 立即超过 $\eta$。故等价性成立。
\end{proof}

\begin{proof}[\normalfont\bfseries 引理~\ref{lem:model:balance}的证明]
由式\eqref{eq:model:full}的第二式，
\[
\dot I=I\left[\frac{\beta c(1-q)S}{N}-\gamma\right].
\]
$I$ 绝对连续且 $I(\tau)=\eta$，故 $I$ 在 $[\tau,\tau']$ 上恒等于 $\eta$，当且仅当 $\dot I$ 在该区间上几乎处处为零。由 $I>0$，这又等价于方括号内几乎处处为零，即式\eqref{eq:model:balance}几乎处处成立。此时，在该区间上几乎处处有 $c(1-q)=\gamma N/(\beta S)>0$，从而 $c>0$ 且 $\dot S=-c[\beta+(1-\beta)q]S\eta/N<0$。任取 $\tau\le u<v\le\tau'$，由 $S$ 的绝对连续性，$S(v)-S(u)=\int_u^v\dot S(t)\,dt<0$，故 $S$ 在该区间上严格递减。
\end{proof}

\subsection{退出后的动态}
\label{sec:proof:post-dynamics}
\begin{proof}[\normalfont\bfseries 定理~\ref{thm:s1:tend}的证明]
(i) 由命题~\ref{prop:model:positive}，$S,I>0$，故恢复常规控制后 $\dot S=-\beta_1^0SI<0$。又由 $S_c=\gamma/\beta_2^0$，$\dot I=\gamma I(S/S_c-1)$，而 $S(t_2)=S_c$，所以对一切 $t>t_2$ 有 $S<S_c$ 和 $\dot I<0$。任取 $t_a>t_2$，令 $\varepsilon=\gamma-\beta_2^0S(t_a)$。由 $S(t_a)<S_c$，$\varepsilon>0$；又 $S$ 递减，故 $t\ge t_a$ 时 $\dot I\le-\varepsilon I$，从而
\[
I(t)\le I(t_a)e^{-\varepsilon(t-t_a)},\qquad t\ge t_a.
\]
因此 $\lim_{t\to\infty}I(t)=0$，且 $\int_{t_a}^{\infty}I(t)\,dt<\infty$。$S$ 单调递减且有下界，极限 $S_\infty$ 存在，且 $S_\infty\le S(t_a)<S_c$。由 $S(t)=S(t_a)\exp[-\beta_1^0\int_{t_a}^tI(s)\,ds]$，得 $S_\infty=S(t_a)\exp[-\beta_1^0\int_{t_a}^{\infty}I(s)\,ds]>0$，故 $S_\infty\in(0,S_c)$。

由常规控制下的首次积分\eqref{eq:model:first-integral}，以 $t_2$ 时的 $(S_c,\eta)$ 为初值，退出后有
\begin{equation}
I(t)=\eta+\frac{\beta_2^0}{\beta_1^0}(S_c-S(t))+\rho_1\ln\frac{S(t)}{S_c}.
\label{eq:s1:post-orbit}
\end{equation}
令 $t\to\infty$，并利用 $\beta_2^0/\beta_1^0=\rho_1/S_c$，得 $x=S_\infty/S_c$ 满足 $(-x)e^{-x}=-e^{-1-\eta/\rho_1}$。由 $0<x<1$，取 $W_0$ 分支，即得式\eqref{eq:s1:Sinfty}。

(ii) 由 (i)，$I$ 连续且从 $\eta>1$ 严格下降至零，故恰有一个 $t_{\rm end}>t_2$ 满足 $I(t_{\rm end})=1$；又 $S(t_{\rm end})<S_c$，故 $\dot I(t_{\rm end})<0$。在式\eqref{eq:s1:post-orbit}中令 $I=1$，同样得 $x=S_{\rm end}/S_c\in(0,1)$ 满足 $(-x)e^{-x}=-e^{-1-(\eta-1)/\rho_1}$，取 $W_0$ 分支即得式\eqref{eq:s1:Send}。最后，由 $\dot S=-\beta_1^0SI$，$t_{\rm end}-t_2=\frac1{\beta_1^0}\int_{S_{\rm end}}^{S_c}\frac{dS}{S\,I(S)}$，其中 $I(S)$ 由式\eqref{eq:s1:post-orbit}给出，即得式\eqref{eq:s1:tend}。
\end{proof}

\subsection{启动状态的导数}
\label{sec:proof:start-sensitivity}
\begin{proof}[\normalfont\bfseries 引理~\ref{lem:s1:Sstar-sensitivity}的证明]
令
\[
F(S;\eta,c_0,\beta,q_0)
=I_0+\frac{\beta_2^0}{\beta_1^0}(S_0-S)+\rho_1\ln(S/S_0)-\eta.
\]
由式\eqref{eq:model:routine-orbit}，$F(S^*)=0$ 即启动条件 $I(S^*)=\eta$。利用 $\beta_2^0/\beta_1^0=\rho_1/S_c$，在 $S=S^*>S_c$ 处
\[
\frac{\partial F}{\partial S}=\rho_1\left(\frac1{S^*}-\frac1{S_c}\right)<0.
\]
因此隐函数定理适用。$\partial F/\partial\eta=-1$。利用 $\beta_2^0/\beta_1^0=\beta(1-q_0)/[\beta+q_0(1-\beta)]$（不含 $c_0$）与 $\rho_1=\gamma N/\{c_0[\beta+q_0(1-\beta)]\}$，固定 $S$ 对各参数求偏导，再在 $S=S^*$ 处取值，得
\begin{align*}
\frac{\partial F}{\partial c_0}
&=-\frac{\rho_1}{c_0}\ln\frac{S^*}{S_0}>0,\\
\frac{\partial F}{\partial\beta}
&=\frac{q_0(1-q_0)}{[\beta+q_0(1-\beta)]^2}(S_0-S^*)
-\frac{\rho_1(1-q_0)}{\beta+q_0(1-\beta)}\ln\frac{S^*}{S_0}>0,\\
\frac{\partial F}{\partial q_0}
&=-\frac{\beta}{[\beta+q_0(1-\beta)]^2}\left[(S_0-S^*)-\bar S\ln\frac{S_0}{S^*}\right]<0.
\end{align*}
由 $S^*<S_0$，$\partial F/\partial c_0>0$；$\partial F/\partial\beta$ 的第一项非负、第二项严格为正。最后一个符号由 $\ln x<x-1$（$x>1$）与 $\bar S<S^*<S_0$ 得到：$\bar S\ln(S_0/S^*)<\bar S(S_0-S^*)/S^*<S_0-S^*$。记 $\kappa$ 为 $\eta,c_0,\beta,q_0$ 中的任一参数，由
$\partial S^*/\partial\kappa=-(\partial F/\partial\kappa)/(\partial F/\partial S)$ 即得所列符号。特别地，
\begin{equation}
\frac{\partial S^*}{\partial\eta}
=\frac1{\rho_1(1/S^*-1/S_c)},\qquad
\frac{\partial S^*}{\partial c_0}
=\frac{\ln(S^*/S_0)}{c_0(1/S^*-1/S_c)}.
\label{eq:s1:Sstar-eta-sign}
\end{equation}
另外两个偏导数为
\begin{align}
\frac{\partial S^*}{\partial\beta}
&=\frac{\dfrac{q_0(1-q_0)(S_0-S^*)}{[\beta+q_0(1-\beta)]^2}
-\dfrac{\rho_1(1-q_0)}{\beta+q_0(1-\beta)}\ln(S^*/S_0)}
{\rho_1(1/S_c-1/S^*)},\label{eq:s1:Sstar-beta-sign}\\
\frac{\partial S^*}{\partial q_0}
&=-\frac{\beta[(S_0-S^*)-\bar S\ln(S_0/S^*)]}
{[\beta+q_0(1-\beta)]^2\rho_1(1/S_c-1/S^*)}.
\label{eq:s1:Sstar-q0-sign}
\end{align}
固定 $N$ 且使用 $\theta=\eta/N$ 时，有 $\partial S^*/\partial\theta=N\partial S^*/\partial\eta$。
\end{proof}

\subsection{阈值的取舍}
\label{sec:proof:threshold-tradeoff}
\begin{proof}[\normalfont\bfseries 命题~\ref{prop:threshold-tradeoff}的证明]
利用引理~\ref{lem:s1:Sstar-sensitivity}中的 $\partial S^*/\partial\eta<0$，依次确定时间、成本与累计感染对阈值的导数符号。

(i) 式\eqref{eq:s1:t1}的被积函数不含 $\eta$，$\eta$ 只通过下限 $S^*$ 进入。由 $I(S^*)=\eta$，
\[
\frac{\partial t_1}{\partial\eta}
=-\frac1{\beta_1^0 S^*\eta}\frac{\partial S^*}{\partial\eta}>0.
\]
对式\eqref{eq:s1:qmax}求导，得
$\partial q_1/\partial\eta=\gamma N[\beta c_0(S^*)^2]^{-1}\partial S^*/\partial\eta<0$。
对式\eqref{eq:s1:t2}求导，得
\[
\frac{\partial\Delta t}{\partial\eta}
=-\frac{\Delta t}{\eta}
+\frac{N}{c_0\eta(S^*-\bar S)}\frac{\partial S^*}{\partial\eta}<0,
\]
其中两项都为负。

(ii) 仅隔离控制下 $(q_c(S)-q_0)/(1-q_0)=1-S_c/S$，$c=c_0$ 时接触项为零，且由式\eqref{eq:s1:S-analytic}，$dt=-N\,dS/[c_0\eta(S-\bar S)]$，故
\begin{equation}
J_q=\frac{N}{c_0\eta}\int_{S_c}^{S^*}\frac{1-S_c/S}{S-\bar S}\,dS,
\qquad
J=\frac{w_qN}{c_0\eta}\int_{S_c}^{S^*}\frac{(1-S_c/S)^2}{S-\bar S}\,dS.
\label{eq:cost:state}
\end{equation}
被积函数不含 $\eta$ 且在区间内部为正。对式\eqref{eq:cost:state}应用 Leibniz 公式，得
\begin{align*}
\frac{\partial J_q}{\partial\eta}
&=-\frac{J_q}{\eta}
+\frac{N}{c_0\eta}\,\frac{1-S_c/S^*}{S^*-\bar S}\,\frac{\partial S^*}{\partial\eta},\\
\frac{\partial J}{\partial\eta}
&=-\frac{J}{\eta}
+\frac{w_qN}{c_0\eta}\,\frac{(1-S_c/S^*)^2}{S^*-\bar S}\,\frac{\partial S^*}{\partial\eta}.
\end{align*}
由 $J_q,J>0$、$S^*>S_c>\bar S$ 与 $\partial S^*/\partial\eta<0$，每式的两项均严格为负。上述积分的闭式表达式见附录~\ref{app:cost}。

(iii) 对累计新增感染，把式\eqref{eq:s1:post-orbit}中 $I=1$ 的条件对 $\eta$ 求导得
\[
\frac{\partial S_{\rm end}}{\partial\eta}
=-\frac1{\rho_1(1/S_{\rm end}-1/S_c)}<0.
\]
对式\eqref{eq:s1:cumulative}求导，
\begin{equation}
\frac{\partial\Itcum(t_{\rm end})}{\partial\eta}
=\left[\frac{\beta S^*}{S^*-\bar S}-\frac{\beta}{\beta+q_0(1-\beta)}\right]\frac{\partial S^*}{\partial\eta}
-\frac{\beta}{\beta+q_0(1-\beta)}\frac{\partial S_{\rm end}}{\partial\eta}>0.
\label{eq:cumulative-eta}
\end{equation}
由 $\beta S_c/(S_c-\bar S)=\beta/[\beta+q_0(1-\beta)]$，方括号等于 $\beta[S^*/(S^*-\bar S)-S_c/(S_c-\bar S)]$。由 $\frac{d}{dS}\frac{S}{S-\bar S}=-\frac{\bar S}{(S-\bar S)^2}<0$ 及 $S^*>S_c$，方括号严格为负；再由 $\partial S^*/\partial\eta<0$ 与 $\partial S_{\rm end}/\partial\eta<0$，式\eqref{eq:cumulative-eta}右端的两项均严格为正。计算至无限时刻时，终点条件改为式\eqref{eq:s1:post-orbit}中 $I=0$。隐式微分给出 $\partial S_\infty/\partial\eta=-[\rho_1(1/S_\infty-1/S_c)]^{-1}<0$；把式\eqref{eq:cumulative-eta}中的 $S_{\rm end}$ 换成 $S_\infty$ 后，右端仍严格为正。
\end{proof}

\subsection{状态参数化}
\label{sec:proof:representation}
\begin{proof}[\normalfont\bfseries 定理~\ref{thm:joint:representation}的证明]
先由速度界建立时间控制与状态反馈函数的对应，再求时间积分公式和退出时的控制极限。

(i) 由引理~\ref{lem:model:balance}，在阈值控制期上几乎处处有
\begin{equation}
c(t)(1-q(t))S(t)=\frac{\gamma N}{\beta}
=c_0(1-q_0)S_c.
\label{eq:joint:balance}
\end{equation}
由式\eqref{eq:joint:balance}与式\eqref{eq:model:plateau-requirement}中的 $q\ge q_0$、$c\le c_0$，阈值控制期内几乎处处有 $q<1$ 和 $c_0S_c/S(t)\le c(t)\le c_0$。由式\eqref{eq:joint:balance}，易感者方程化为 $\dot S=-(\eta/N)[c(t)S(t)-c_0\bar S]$，于是几乎处处有
\[
\frac{\eta c_0}{N}(S_c-\bar S)\le-\dot S(t)\le\frac{\eta c_0}{N}(S^*-\bar S),
\]
左端为正。由此，$S$ 在阈值控制期上严格递减，$S(t)$ 与其反函数 $t(S)$ 均为 Lipschitz 函数，零测集在两个变量之间的变换下仍为零测集。因此，与 $S(t)$ 或 $t(S)$ 复合均保持可测性及几乎处处相等关系。令 $c_c(S)=c(t(S))$，则 $c_c$ 可测，$c(t)=c_c(S(t))$，$c_c$ 满足式\eqref{eq:joint:contact-range}，式\eqref{eq:joint:balance}给出式\eqref{eq:joint:q}。

(ii) 由式\eqref{eq:joint:contact-range}，$c_c(S)S\in[c_0S_c,c_0S]$，式\eqref{eq:joint:Sdot}满足 (i) 中的速度界。因此 $S$ 在有限时间内从 $S^*$ 降至 $S_c$；分离变量即得式\eqref{eq:joint:time}和\eqref{eq:joint:duration}。

反过来，给定满足式\eqref{eq:joint:contact-range}的可测函数 $c_c$，可在零测集上修改使界限处处成立，经式\eqref{eq:joint:q}给出的 $q_c$ 满足 $q_0\le q_c<1$。式\eqref{eq:joint:time}右侧随积分下限严格递减，由速度界，它唯一确定绝对连续、严格递减的 $S(t)$。用式\eqref{eq:joint:q}构造时间控制 $c(t)=c_c(S(t))$、$q(t)=q_c(S(t))$，令 $I\equiv\eta$，则 $S$ 与 $I$ 的方程几乎处处成立，其余仓室由原模型的方程确定；由命题~\ref{prop:model:positive}中解的唯一性，这就是该控制下的解。$S$ 在 $t_2$ 首次降到 $S_c$，该控制满足定义~\ref{def:model:threshold-strategy}，属于 $\mathcal A(\eta)$。因此 (i) 中的对应在几乎处处相等的意义下成立。

(iii) 取处处满足式\eqref{eq:joint:contact-range}的代表。由 (ii)，$S$ 连续且严格递减，$S(t_2)=S_c$，故 $t_1\le t<t_2$ 时 $S(t)>S_c$，且 $\lim_{t\to t_2^{-}}S(t)=S_c$。又 $c_0S_c/S(t)\le c(t)\le c_0$，由夹逼得 $\lim_{t\to t_2^{-}}c(t)=c_0$；再由式\eqref{eq:joint:q}，$\lim_{t\to t_2^{-}}q(t)=q_0$。若 $c_c$ 连续，则 $c(t)=c_c(S(t))$ 与由式\eqref{eq:joint:q}给出的 $q(t)$ 都是连续函数的复合，在 $(t_1,t_2)$ 上连续。
\end{proof}

\subsection{最小值的存在性}
\label{sec:proof:cost-minimum}
\begin{proof}[\normalfont\bfseries 定理~\ref{thm:joint:cost-minimum}的证明]
(i) 允许区间上 $c\ge c_0S_c/S$，故式\eqref{eq:joint:cost-integrand}的分母满足
$cS-c_0\bar S\ge c_0(S_c-\bar S)>0$，$f$ 在允许区域上连续。每个状态上的允许区间紧且非空，故逐状态最小值存在，任意允许控制的成本均不小于式\eqref{eq:joint:minimum-cost}右侧。为证明这一下界可以达到，将允许区间写为
$c=c_0[S_c/S+x(1-S_c/S)]$，$x\in[0,1]$，并记
\[
H(S,x)=f\!\left(S,c_0\Bigl[\frac{S_c}{S}+x\Bigl(1-\frac{S_c}{S}\Bigr)\Bigr]\right),
\quad m(S)=\min_{0\le x\le1}H(S,x).
\]
$H$ 在紧集 $[S_c,S^*]\times[0,1]$ 上连续，因而一致连续；对 $x$ 取最小值后，$m$ 关于 $S$ 连续。每个状态上的最小点集均为非空紧集，取其最小元素 $x_*(S)$。对任意 $0<a\le1$，
\[
\{S:x_*(S)<a\}
=\bigcup_{\substack{a'\in\mathbb Q\\0\le a'<a}}
\left\{S:\min_{0\le x\le a'}H(S,x)=m(S)\right\}.
\]
对每个固定的 $a'$，函数 $S\mapsto\min_{0\le x\le a'}H(S,x)$ 连续，故右侧为可数个闭集的并，$x_*$ 因而可测。于是
$c_c^*(S)=c_0[S_c/S+x_*(S)(1-S_c/S)]$
是可测的逐状态最小化选择。由定理~\ref{thm:joint:representation}得到相应的允许控制，其成本达到上述下界。

(ii) 对任意允许控制，$f(S,c_c(S))-m(S)\ge0$。因此 $J[c_c]-J_{\min}=0$，当且仅当 $f(S,c_c(S))=m(S)$ 对几乎所有 $S$ 成立。

(iii) 固定 $S$，对式\eqref{eq:joint:cost-integrand}关于 $c$ 求偏导，并乘以正数
$c^3(cS-c_0\bar S)^2/S$，驻点方程化为
\begin{equation}
\begin{split}
0={}&w_c\frac{c^3}{c_0^2}(c-c_0)
\left(c+c_0-\frac{2c_0\bar S}{S}\right)\\
&+w_q\left(c-\frac{c_0S_c}{S}\right)
\left(-c^2+\frac{3c_0S_c}{S}c-
\frac{2c_0^2\bar S S_c}{S^2}\right).
\end{split}
\label{eq:joint:stationary}
\end{equation}
该多项式最高次系数为 $w_c/c_0^2>0$，至多有五个实根。连续可微函数在闭区间上的最小点只能是端点或内部驻点。
\end{proof}

\subsection{退出附近的控制组合}
\label{sec:proof:terminal-allocation}
\begin{proof}[\normalfont\bfseries 命题~\ref{prop:joint:terminal-allocation}的证明]
先由归一化后的被积函数的一致收敛确定退出附近的最优组合，再比较两种单控制的成本。

(i) 令 $\epsilon=1-S_c/S$，则 $S>S_c$ 时 $0<\epsilon<1$，且 $\lim_{S\to S_c^{+}}\epsilon=0$。把允许接触率写为
$c=c_0(1-\epsilon x)$、$0\le x\le1$，于是
\[
\frac{q_c-q_0}{1-q_0}=\frac{\epsilon(1-x)}{1-\epsilon x},\qquad
\frac{f(S,c_0(1-\epsilon x))}{\epsilon^2}
=\frac{w_cx^2+w_q(1-x)^2/(1-\epsilon x)^2}
{c_0[S(1-\epsilon x)-\bar S]}.
\]
记上式右侧为 $g_S(x)$，并记
\[
g(x)=\frac{w_cx^2+w_q(1-x)^2}{c_0(S_c-\bar S)},
\]
由 $1-\epsilon x\ge S_c/S$ 与 $S(1-\epsilon x)-\bar S\ge S_c-\bar S>0$，$g_S$ 的分母一致远离零，因而 $\lim_{S\to S_c^{+}}\sup_{0\le x\le1}|g_S(x)-g(x)|=0$。$g$ 严格凸，唯一最小点为 $x^*=w_q/(w_c+w_q)$。记 $X(S)$ 为 $g_S$ 在 $[0,1]$ 上的最小点集合，下证
\[
\lim_{S\to S_c^{+}}\sup_{x\in X(S)}|x-x^*|=0.
\]
否则存在 $S_n>S_c$ 与 $x_n\in X(S_n)$，使 $\lim_{n\to\infty}S_n=S_c$，而 $|x_n-x^*|$ 有正下界。取子列使 $x_n\to\bar x$，则 $\bar x\ne x^*$。由 $g_{S_n}(x_n)\le g_{S_n}(x^*)$ 及一致收敛，得 $g(\bar x)\le g(x^*)$，与最小点的唯一性矛盾。因此存在 $\delta>0$，使 $S_c<S<S_c+\delta$ 时所有最小点都位于 $x\in(0,1)$，即得式\eqref{eq:joint:terminal-interior}。又由 $(c_0-c)/c_0=\epsilon x$ 与上面第一式，式\eqref{eq:joint:terminal-ratio}中的两个比值分别等于 $x$ 与 $(1-x)/(1-\epsilon x)$，令 $S$ 从右侧趋于 $S_c$ 即得式\eqref{eq:joint:terminal-ratio}。

(ii) 在长度为正的区间 $(S_c,\min\{S_c+\delta,S^*\})$ 上，两种单控制分别对应 $x=0$ 和 $x=1$，两种单控制对应的 $f(S,c)$ 均严格大于该状态上的最小值；在其余状态上，成本差非负。因此，两种单控制的总成本均严格大于最小成本。
\end{proof}

\subsection{连续结构}
\label{sec:proof:cost-structure}
令 $\zeta=S/S_c$、$p=cS/(c_0S_c)$，则允许区间 $[c_0S_c/S,c_0]$ 对应 $p\in[1,\zeta]$：$p=1$ 为仅减少接触，$p=\zeta$ 为仅隔离控制。由 $c/c_0=p/\zeta$、$(q-q_0)/(1-q_0)=1-1/p$ 与 $cS-c_0\bar S=c_0S_c(p-b)$，
\[
f(S,c)=\frac{w_c}{c_0S_c}\,G(p;\zeta),\qquad
G(p;\zeta)=\frac{(1-p/\zeta)^2+r(1-1/p)^2}{p-b}.
\]
固定 $S$ 时 $p$ 是 $c$ 的严格增函数，只需讨论 $G(\cdot;\zeta)$。记 $A(p;\zeta)=(1-p/\zeta)^2/(p-b)$、$B(p)=(1-1/p)^2/(p-b)\ge0$，则 $G(p;\zeta)=A(p;\zeta)+rB(p)$。再记 $\zeta_{\rm sw}=S_{\rm sw}/S_c=(3+\sqrt{9-8b})/2\in(2,3)$。直接计算得 $G_p=E/(p-b)^2$，其中
\[
E(p,\zeta)=\frac{p^2-2bp}{\zeta^2}+\frac{2b}{\zeta}-1+r\varphi(p),\qquad
\varphi(p)=\frac{(p-1)(3p-p^2-2b)}{p^3},
\]
并且
\begin{equation}
E_p=2(p-b)\Bigl[\frac1{\zeta^2}-r\psi(p)\Bigr],\quad
\psi(p)=\frac{2p-3}{p^4};\qquad
E_\zeta=-\frac{2[(p-b)^2+b(\zeta-b)]}{\zeta^3}<0.
\label{eq:proof:cost-E}
\end{equation}
两端的值为 $E(1,\zeta)=-(\zeta-1)(\zeta+1-2b)/\zeta^2<0$ 与 $E(\zeta,\zeta)=r\varphi(\zeta)$。后者在 $1<\zeta<\zeta_{\rm sw}$ 时为正，在 $\zeta=\zeta_{\rm sw}$ 时为零，在 $\zeta>\zeta_{\rm sw}$ 时为负（命题~\ref{prop:joint:global-structure}）。下面的引理用于证明内部驻点曲线单调。
\begin{lemma}
\label{lem:proof:fold-free}
设 $0<b<1$，$\sigma_c=\sqrt{r_{\rm cont}}$。对 $p>3/2$ 与 $\sigma>0$，记
\[
F(p;\sigma)=\sigma^2X(p)-2\sigma bp\sqrt{2p-3}+p^3,\qquad X(p)=p^3-6p^2+6p+2b(3p-4).
\]
则 $0<\sigma<\sigma_c$ 时 $F(p;\sigma)>0$；$\sigma=\sigma_c$ 时 $F(p;\sigma_c)\ge0$，且等号仅在 $p=2$ 成立。
\end{lemma}
\begin{proof}
先证 $\sigma=\sigma_c$ 时的非负性与等号条件，再由 $F/\sigma^2$ 关于 $\sigma$ 的单调性得到 $0<\sigma<\sigma_c$ 时的严格正性。

由式\eqref{eq:joint:rcont}，$\sigma_c=4/(b+\sqrt{b^2-8b+8})\in(\sqrt2,2)$ 满足 $(1-b)\sigma_c^2+b\sigma_c-2=0$，即 $b=(\sigma_c^2-2)/[\sigma_c(\sigma_c-1)]$。令 $w=\sqrt{2p-3}$，代入并展开得
\[
F(p;\sigma_c)=\frac{(w-1)^2\,Q(w,\sigma_c)}{8(\sigma_c-1)},
\]
\[
\begin{split}
Q(w,\sigma)={}&(\sigma-1)(\sigma^2+1)w^3(w+2)+12(\sigma-1)w^2\\
&-2(\sigma^3+3\sigma^2-11\sigma+3)w+(9\sigma^2+11\sigma-\sigma^3-27).
\end{split}
\]
在 $\sigma\in(\sqrt2,2)$ 上逐项估计 $Q$ 的系数。常数项 $9\sigma^2+11\sigma-\sigma^3-27$ 的导数 $18\sigma+11-3\sigma^2$ 为正，故常数项随 $\sigma$ 增，不小于其在 $\sigma=\sqrt2$ 处的值 $9(\sqrt2-1)$。一次项系数为 $-2(\sigma^3+3\sigma^2-11\sigma+3)$，括号内的导数 $3\sigma^2+6\sigma-11$ 为正，故该系数随 $\sigma$ 减，不小于其在 $\sigma=2$ 处的值 $-2$。二次项系数 $12(\sigma-1)$ 大于 $12(\sqrt2-1)$，三、四次项非负。所以对 $w\ge0$ 有 $Q\ge9(\sqrt2-1)-2w+12(\sqrt2-1)w^2>0$（判别式为负），从而 $F(p;\sigma_c)\ge0$，等号仅在 $w=1$ 即 $p=2$ 成立。

下证 $0<\sigma<\sigma_c$ 的情形。由 $p^4-16(2p-3)=(p-2)^2(p^2+4p+12)\ge0$，$p^2\ge4w$；又 $0<b<1$、$\sigma_c<2$，故 $0<\sigma\le\sigma_c$ 时 $bw\sigma<p^2$，从而
\[
\frac{\partial}{\partial\sigma}\frac{F(p;\sigma)}{\sigma^2}
=\frac{2p(bw\sigma-p^2)}{\sigma^3}<0.
\]
因此 $0<\sigma<\sigma_c$ 时 $F(p;\sigma)/\sigma^2>F(p;\sigma_c)/\sigma_c^2\ge0$，即 $F(p;\sigma)>0$。
\end{proof}

\begin{proof}[\normalfont\bfseries 定理~\ref{thm:joint:cost-structure}的证明]
先刻画内部驻点曲线 $\zeta=\zeta_s(p)$，再由引理~\ref{lem:proof:fold-free}证明它严格递增，最后确定各状态上的唯一最小点及其连续性。

\emph{第一步：内部驻点曲线。}由 $\varphi'(p)=-2(2p-3)(p-b)/p^4$，$\varphi$ 在 $(1,\zeta_{\rm sw})$ 上为正，最大值为 $\varphi(3/2)=(9-8b)/27$，在 $p\ge\zeta_{\rm sw}$ 时不为正。又 $b^2-8b+8=2(2-b)^2-b^2$，展开得 $(b+\sqrt{b^2-8b+8})^2=2(2-b)^2+2b\sqrt{b^2-8b+8}\ge2(2-b)^2$，故 $r_{\rm cont}\le8/(2-b)^2$；而 $27(2-b)^2-8(9-8b)=27b^2-44b+36>0$（判别式为负），故 $r_{\rm cont}<27/(9-8b)$。设 $r\le r_{\rm cont}$，则对所有 $p>1$ 有 $r\varphi(p)<1$。固定 $p>1$，由式\eqref{eq:proof:cost-E}，$E(p,\cdot)$ 在 $[p,\infty)$ 上严格减，且
\[
E(p,p)=r\varphi(p),\qquad
\lim_{\zeta\to\infty}E(p,\zeta)=r\varphi(p)-1<0.
\]
因此满足 $1<p<\zeta$、$E(p,\zeta)=0$ 的点恰好组成一条曲线 $\zeta=\zeta_s(p)$，$p\in(1,\zeta_{\rm sw})$。由隐函数定理，$\zeta_s$ 连续可微，且 $\zeta_s'=-E_p/E_\zeta$，所以沿曲线 $\zeta_s'$ 与 $E_p$ 同号。再求曲线的端点极限。$E(1,1)=E(\zeta_{\rm sw},\zeta_{\rm sw})=0$，且
\[
E_\zeta(1,1)=-2(1-b)<0,\qquad
E_\zeta(\zeta_{\rm sw},\zeta_{\rm sw})=-\frac{2(\zeta_{\rm sw}-b)}{\zeta_{\rm sw}^2}<0.
\]
由隐函数定理，在这两点附近 $E=0$ 的解构成连续曲线 $\zeta=g(p)$，分别满足 $g(1)=1$ 与 $g(\zeta_{\rm sw})=\zeta_{\rm sw}$。对端点附近的 $p\in(1,\zeta_{\rm sw})$，$E(p,p)=r\varphi(p)>0$ 而 $E(p,\cdot)$ 严格减，故 $g(p)>p$；由 $(p,\infty)$ 内零点的唯一性，$g(p)=\zeta_s(p)$。因此
\[
\lim_{p\to1^{+}}\zeta_s(p)=1,\qquad
\lim_{p\to\zeta_{\rm sw}^{-}}\zeta_s(p)=\zeta_{\rm sw}.
\]

\emph{第二步：曲线单调。}$p\le3/2$ 时 $\psi(p)\le0$，故 $E_p>0$。若曲线上某点 $p>3/2$ 处 $E_p=0$，则 $1/\zeta=\sqrt{r\psi(p)}$；代入 $E=0$ 并乘以 $-p^3$，得 $F(p;\sqrt r)=0$。先设 $r<r_{\rm cont}$。由引理~\ref{lem:proof:fold-free}，曲线上处处 $E_p\ne0$；而 $p\le3/2$ 时 $E_p>0$，由 $E_p$ 沿曲线连续，曲线上处处 $E_p>0$。再设 $r=r_{\rm cont}$。由引理~\ref{lem:proof:fold-free}，曲线上 $E_p$ 的零点只可能在 $p=2$。由隐函数定理及 $E_\zeta<0$，对每个固定的 $p$，$\zeta_s(p)$ 随 $r$ 连续变化；在 $r<r_{\rm cont}$ 时的不等式 $E_p(p,\zeta_s(p))>0$ 中令 $r$ 从左侧趋于 $r_{\rm cont}$，得曲线上 $E_p\ge0$，因而 $\zeta_s'\ge0$，且至多在 $p=2$ 处为零。于是对 $1<p_1<p_2<\zeta_{\rm sw}$，$\zeta_s(p_2)-\zeta_s(p_1)=\int_{p_1}^{p_2}\zeta_s'(p)\,dp>0$。结合第一步的两个端点极限，$\zeta_s$ 在两种情形下都是 $(1,\zeta_{\rm sw})$ 到自身的严格递增连续双射。

\emph{第三步：最小点及其连续性。}对 $1<\zeta<\zeta_{\rm sw}$，$E(\cdot,\zeta)$ 在 $(1,\zeta)$ 内恰有一个零点 $p^*(\zeta)=\zeta_s^{-1}(\zeta)$；又 $E(1,\zeta)<0<E(\zeta,\zeta)$，所以 $E$ 在 $p^*$ 处由负变正，$G(\cdot;\zeta)$ 先严格减、后严格增，最小点唯一且在内部。对 $\zeta\ge\zeta_{\rm sw}$，$E(\cdot,\zeta)$ 在 $(1,\zeta)$ 内没有零点且 $E(1,\zeta)<0$，故在整个内部区间上 $G_p<0$。因此，$G$ 在 $[1,\zeta]$ 上严格递减，唯一最小点为 $p=\zeta$。由第二步，$p^*$ 是 $(1,\zeta_{\rm sw})$ 到自身的严格增连续双射，故
\[
\lim_{\zeta\to1^{+}}p^*(\zeta)=1,\qquad
\lim_{\zeta\to\zeta_{\rm sw}^{-}}p^*(\zeta)=\zeta_{\rm sw}.
\]
回到原变量，$S_c<S<S_{\rm sw}$ 时 $c_c^*(S)=c_0S_cp^*(S/S_c)/S$，$S\ge S_{\rm sw}$ 时 $c_c^*(S)=c_0$。由上述两个极限，
\[
\lim_{S\to S_c^{+}}c_c^*(S)=\lim_{S\to S_{\rm sw}^{-}}c_c^*(S)=c_0,
\]
故 $c_c^*$ 在 $(S_c,\infty)$ 上连续，即得 (i)(ii)。由定理~\ref{thm:joint:cost-minimum}(ii)，成本最优控制几乎处处等于 $c_c^*$，再由 (ii) 即得 (iii)。
\end{proof}

\subsection{权重单调性}
\label{sec:proof:weight-monotone}
沿用上一小节的记号。
\begin{proof}[\normalfont\bfseries 命题~\ref{prop:joint:weight-monotone}的证明]
(i) 由 $A_p=-(\zeta-p)(\zeta+p-2b)/[\zeta^2(p-b)^2]<0$（$1\le p<\zeta$），$A$ 在 $[1,\zeta]$ 上严格递减。固定 $S$，设 $p_1$、$p_2$ 分别是 $r_1<r_2$ 下的最小点，二者属于同一允许区间。由最优性，
\[
\begin{aligned}
A(p_1)+r_1B(p_1)&\le A(p_2)+r_1B(p_2),\\
A(p_2)+r_2B(p_2)&\le A(p_1)+r_2B(p_1).
\end{aligned}
\]
两式相加得 $(r_2-r_1)[B(p_2)-B(p_1)]\le0$，即 $B(p_2)\le B(p_1)$。若 $p_2>p_1$，则 $A(p_2)<A(p_1)$，于是 $A(p_2)+r_1B(p_2)<A(p_1)+r_1B(p_1)$，与 $p_1$ 在 $r_1$ 下最优矛盾。故 $p_2\le p_1$。固定 $S$ 时，$c$ 与 $q$ 均随 $p$ 严格递增，因此两者的最优取值均随 $r$ 不增。

由式\eqref{eq:joint:duration}，$c_{r_2}^*\le c_{r_1}^*$ 给出相应的控制时长不减。式\eqref{eq:joint:cumulative}与式\eqref{eq:joint:quarantine}中 $\Delta t$ 的系数分别为正和负，故阈值控制期累计新增感染不减，新增隔离易感者人数不增。

对任一允许控制，$J/w_c$ 是 $r$ 的仿射函数，其斜率为 $\int_{t_1}^{t_2}u_q(t)^2\,dt\ge0$。仿射函数族的下确界为凹函数，而这族函数都不减，故 $J_{\min}/w_c$ 关于 $r$ 为凹函数且不减。

(ii) 由式\eqref{eq:proof:cost-E}，$G_{p\zeta}=E_\zeta/(p-b)^2<0$。设 $\zeta_1<\zeta_2$，任取 $\zeta_i$ 下的最小点 $p_i$，$i=1,2$。反设 $p_2<p_1$。由 $1\le p_2<p_1\le\zeta_1<\zeta_2$，$p_1$、$p_2$ 均属于两个状态的允许区间。由最优性，
\[
[G(p_1;\zeta_2)-G(p_2;\zeta_2)]-[G(p_1;\zeta_1)-G(p_2;\zeta_1)]\ge0,
\]
而左侧等于 $\int_{\zeta_1}^{\zeta_2}\int_{p_2}^{p_1}G_{p\zeta}\,dp\,d\zeta<0$，矛盾。上述比较对任意两个最小点成立，故任一逐状态最小化选择对应的 $p$ 随 $\zeta$ 不减。由 $q=1-(1-q_0)/p$，相应的隔离比例随 $S$ 不减；又 $S(t)$ 严格递减，故 $q_r^*(S(t))$ 随时间不增。
\end{proof}
