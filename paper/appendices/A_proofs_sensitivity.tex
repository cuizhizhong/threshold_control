\subsection{解的适定性}
\label{sec:proof:model-positive}
本小节证明命题\ref{prop:model:positive}。
\begin{proof}
方程右端关于状态局部 Lipschitz，关于时间可测且在有界状态集上局部可积，故有唯一局部绝对连续解。前两式给出
\begin{align*}
S(t)&=S_0\exp\!\left[-\int_0^t\frac{c(s)[\beta+(1-\beta)q(s)]}{N}I(s)\,ds\right],\\
I(t)&=I_0\exp\!\left[\int_0^t\left(\frac{\beta c(s)(1-q(s))}{N}S(s)-\gamma\right)ds\right],
\end{align*}
由此得 $S,I$ 的严格正性。$S_q$ 的流入非负，$I_q$ 由常数变易公式保持非负，$\dot R\ge0$。五式相加得总人数守恒，因此各分量有界，局部解可以延拓至任意有限时刻。
\end{proof}

\subsection{触发与退出}
\label{sec:proof:threshold-rationale}
本小节证明命题\ref{prop:model:threshold-rationale}。
\begin{proof}
(i) 常规轨迹在首次到达时满足
\[
\dot I(t_1)=\eta\left[\frac{\beta c_0(1-q_0)S^*}{N}-\gamma\right]>0.
\]
若其后仍维持常规控制一段正时间，由连续性 $I$ 将超过 $\eta$，故首次加强时刻不能晚于 $t_1$；$[0,t_1]$ 上 $I\le\eta$ 由首次到达的定义得到。

(ii) 由感染者方程在 $I=\eta$ 处直接得到。对有界可测控制，若 $I$ 在一个区间上恒等于 $\eta$，则几乎处处 $\dot I=0$，故等号几乎处处成立。

(iii) 恢复常规控制后
\[
\dot I=\gamma I\left(\frac{S}{S_c}-1\right),\qquad
\dot S=-\frac{c_0[\beta+(1-\beta)q_0]}{N}SI<0.
\]
若恢复时 $S\le S_c$，此后 $S$ 严格下降，所有后续时刻都有 $\dot I<0$，$I$ 不再超过 $\eta$。若恢复时 $S>S_c$，则恢复后的一个右邻域内 $\dot I>0$，立即违反 $I\le\eta$。阈值控制期内 $S$ 也严格下降，故沿该轨迹首次到达 $S_c$ 时即可恢复常规控制，在此之前则不能从阈值上直接退出。
\end{proof}

\subsection{退出后的动态}
\label{sec:proof:post-dynamics}
本小节证明定理\ref{thm:s1:tend}。
\begin{proof}
由正性，恢复常规后 $\dot S<0$，故所有 $t>t_2$ 均有 $S<S_c$ 和 $\dot I<0$。任取 $t_a>t_2$，令 $\varepsilon=\gamma-\beta_2^0S(t_a)>0$，则
\[
I(t)\le I(t_a)e^{-\varepsilon(t-t_a)},\qquad t\ge t_a.
\]
因此 $I\to0$、$\int_{t_a}^{\infty}I(t)dt<\infty$。将其代入 $S$ 的指数表示，得到严格正的下界，故 $S_\infty\in(0,S_c)$。$I$ 从 $\eta>1$ 严格下降至零，故存在唯一有限的 $t_{\rm end}$。

退出后的首次积分为
\begin{equation}
I(t)=\eta+\frac{\beta_2^0}{\beta_1^0}(S_c-S(t))+\rho_1\ln\frac{S(t)}{S_c}.
\label{eq:s1:post-orbit}
\end{equation}
分别令 $I=1$ 和 $I=0$，并使用 $0<S<S_c$ 选择 $W_0$ 分支，得到式\eqref{eq:s1:Send}和\eqref{eq:s1:Sinfty}。最后沿常规易感者方程积分即得式\eqref{eq:s1:tend}。
\end{proof}

\subsection{启动状态的导数}
\label{sec:proof:start-sensitivity}
本小节证明引理\ref{lem:s1:Sstar-sensitivity}。
\begin{proof}
令
\[
F(S;\eta,c_0,\beta,q_0)
=I_0+\frac{\beta_2^0}{\beta_1^0}(S_0-S)+\rho_1\ln(S/S_0)-\eta.
\]
在 $S=S^*>S_c$ 处，
\[
\frac{\partial F}{\partial S}=\rho_1\left(\frac1{S^*}-\frac1{S_c}\right)<0.
\]
因此隐函数定理适用。$\partial F/\partial\eta=-1$，而在固定 $S=S^*$ 时
\begin{align*}
\frac{\partial F}{\partial c_0}
&=-\frac{\rho_1}{c_0}\ln\frac{S^*}{S_0}>0,\\
\frac{\partial F}{\partial\beta}
&=\frac{q_0(1-q_0)}{[\beta+q_0(1-\beta)]^2}(S_0-S^*)
-\frac{\rho_1(1-q_0)}{\beta+q_0(1-\beta)}\ln\frac{S^*}{S_0}>0,\\
\frac{\partial F}{\partial q_0}
&=-\frac{\beta}{[\beta+q_0(1-\beta)]^2}\left[(S_0-S^*)-\bar S\ln\frac{S_0}{S^*}\right]<0.
\end{align*}
最后一个符号使用 $\bar S<S^*<S_0$ 及 $\ln x<x-1$。由
$\partial S^*/\partial p=-(\partial F/\partial p)/(\partial F/\partial S)$ 即得所列符号。特别地，
\begin{equation}
\frac{\partial S^*}{\partial\eta}
=\frac1{\rho_1(1/S^*-1/S_c)},\qquad
\frac{\partial S^*}{\partial c_0}
=\frac{\ln(S^*/S_0)}{c_0(1/S^*-1/S_c)}.
\label{eq:s1:Sstar-eta-sign}
\end{equation}
另两个偏导数的完整表达为
\begin{align}
\frac{\partial S^*}{\partial\beta}
&=\frac{\dfrac{q_0(1-q_0)(S_0-S^*)}{[\beta+q_0(1-\beta)]^2}
-\dfrac{\rho_1(1-q_0)}{\beta+q_0(1-\beta)}\ln(S^*/S_0)}
{\rho_1(1/S_c-1/S^*)},\label{eq:s1:Sstar-beta-sign}\\
\frac{\partial S^*}{\partial q_0}
&=-\frac{\beta[(S_0-S^*)-\bar S\ln(S_0/S^*)]}
{[\beta+q_0(1-\beta)]^2\rho_1(1/S_c-1/S^*)}.
\label{eq:s1:Sstar-q0-sign}
\end{align}
固定 $N$ 且使用 $\theta=\eta/N$ 时，有 $\partial S^*/\partial\theta=N\partial S^*/\partial\eta$。
\end{proof}

\subsection{阈值的取舍}
\label{sec:proof:threshold-tradeoff}
本小节证明命题\ref{prop:threshold-tradeoff}。
\begin{proof}
由式\eqref{eq:s1:t1}及 $I(S^*)=\eta$，
\[
\frac{\partial t_1}{\partial\eta}
=-\frac1{\beta_1^0 S^*\eta}\frac{\partial S^*}{\partial\eta}>0.
\]
式\eqref{eq:s1:qmax}给出
$\partial q_{\max}/\partial\eta=\gamma N[\beta c_0(S^*)^2]^{-1}\partial S^*/\partial\eta<0$。
控制时长满足
\[
\frac{\partial\Delta t}{\partial\eta}
=-\frac{\Delta t}{\eta}
+\frac{N}{c_0\eta(S^*-\bar S)}\frac{\partial S^*}{\partial\eta}<0.
\]
又由于 $(q_c(S)-q_0)/(1-q_0)=1-S_c/S$，有
\begin{equation}
J_q=\frac{N}{c_0\eta}\int_{S_c}^{S^*}\frac{1-S_c/S}{S-\bar S}\,dS,
\qquad
J=\frac{2N}{c_0\eta}\int_{S_c}^{S^*}\frac{(1-S_c/S)^2}{S-\bar S}\,dS.
\label{eq:cost:state}
\end{equation}
被积函数不含 $\eta$ 且在区间内部为正，系数 $1/\eta$ 和上限 $S^*$ 均随 $\eta$ 增大而减小，故两个成本严格下降。两积分的初等表达列于附录\ref{app:cost}。

对累计新增感染，把退出后 $I=1$ 的条件对 $\eta$ 求导得
\[
\frac{\partial S_{\rm end}}{\partial\eta}
=-\frac1{\rho_1(1/S_{\rm end}-1/S_c)}<0.
\]
对式\eqref{eq:s1:cumulative}求导，
\begin{equation}
\frac{\partial\Itcum}{\partial\eta}
=\left[\frac{\beta S^*}{S^*-\bar S}-\frac{\beta}{\beta+q_0(1-\beta)}\right]\frac{\partial S^*}{\partial\eta}
-\frac{\beta}{\beta+q_0(1-\beta)}\frac{\partial S_{\rm end}}{\partial\eta}>0.
\label{eq:cumulative-eta}
\end{equation}
其中使用 $\beta S_c/(S_c-\bar S)=\beta/[\beta+q_0(1-\beta)]$，以及 $S/(S-\bar S)$ 在 $S>\bar S$ 上非增。因此方括号非正，其与 $\partial S^*/\partial\eta<0$ 的乘积非负，最后一项严格为正。将终点替换为 $I=0,S=S_\infty$，同样的隐式微分给出无限时刻的结论。
\end{proof}

\subsection{人口缩放}
\label{sec:proof:scaling}
本小节证明命题\ref{prop:scaling}。
\begin{proof}
常规阶段的归一化系统为
\[
\dot s=-c_0[\beta+q_0(1-\beta)]si,\qquad
\dot i=[\beta c_0(1-q_0)s-\gamma]i.
\]
它不含 $N$，触发条件为 $i=\theta$，所以 $t_1$ 与 $s^*=S^*/N$ 仅依赖上述固定量和 $\theta$。又
\[
s_c=\frac{\gamma}{\beta c_0(1-q_0)},\qquad
\bar s=\frac{(1-\beta)\gamma}{\beta c_0},\qquad
\Delta t=\frac1{c_0\theta}\ln\frac{s^*-\bar s}{s_c-\bar s}.
\]
例如，二次干预强度成本的归一化表达为
\begin{equation}
J=\frac{2}{c_0\theta}\int_{s_c}^{s^*}
\frac{(1-s_c/s)^2}{s-\bar s}\,ds,
\label{eq:dom:Jtheta}
\end{equation}
其中只含归一化状态和固定参数。代入控制表达可得其余结论。无限时间的终点为 $i\to0$，同样不含 $N$；有限的终点 $I=1$ 却对应 $i=1/N$，因而保留绝对人口尺度。
\end{proof}

\subsection{状态参数化}
\label{sec:proof:representation}
本小节证明定理\ref{thm:joint:representation}。
\begin{proof}
在 $I=\eta>0$ 上，感染者方程给出
\begin{equation}
c(t)(1-q(t))S(t)=\frac{\gamma N}{\beta}
=c_0(1-q_0)S_c.
\label{eq:joint:balance}
\end{equation}
因此在 $I=\eta>0$ 上 $q<1$。由 $q\ge q_0$ 和 $c\le c_0$，可得式\eqref{eq:joint:contact-range}和\eqref{eq:joint:q}；反向代入也成立。利用式\eqref{eq:joint:balance}，易感者方程化为式\eqref{eq:joint:Sdot}，其下降速度满足
\[
0<c_0(S_c-\bar S)\le c_c(S)S-c_0\bar S
\le c_0(S^*-\bar S).
\]
因此 $S$ 在有限时间从 $S^*$ 到达 $S_c$，分离变量即得式\eqref{eq:joint:time}和\eqref{eq:joint:duration}。

反之，给定满足式\eqref{eq:joint:contact-range}的可测函数，可在零测集上修改使界限处处成立。式\eqref{eq:joint:time}右侧随积分下限严格递减，且该时间变换及其逆函数均为 Lipschitz 函数，故定义唯一绝对连续的 $S(t)$。用式\eqref{eq:joint:q}构造时间控制，令 $I=\eta$，则前两个状态方程几乎处处成立；其余仓室由原模型确定。该时间变换及其逆都是 Lipschitz 的，保持零测集与可测性，所以这一对应在几乎处处相等意义下成立。最后，由 $c_0S_c/S\le c_c(S)\le c_0$，当 $S\downarrow S_c$ 时两界均趋于 $c_0$；式\eqref{eq:joint:q}继而给出 $q\to q_0$。
\end{proof}

\subsection{最小值的存在性}
\label{sec:proof:cost-minimum}
本小节证明定理\ref{thm:joint:cost-minimum}。
\begin{proof}
式\eqref{eq:joint:cost}的分母满足
$c_c(S)S-c_0\bar S\ge c_0(S_c-\bar S)>0$，所以 $f$ 在允许区域上连续有限。每个状态上的允许区间紧且非空，故最小值存在。任意允许控制的成本均不小于式\eqref{eq:joint:minimum-cost}右侧。

为证明下界可以达到，将允许区间写为
$c=c^{-}(S)+x[c^{+}(S)-c^{-}(S)]$，$x\in[0,1]$，并记
\[
H(S,x)=f\!\left(S,c^{-}(S)+x[c^{+}(S)-c^{-}(S)]\right),
\quad m(S)=\min_{0\le x\le1}H(S,x).
\]
$H$ 在紧集上连续，因而 $m$ 连续。取各最小化解集中最小的元素 $x_*(S)$。对于任意 $0<a\le1$，
\[
\{S:x_*(S)<a\}
=\bigcup_{\substack{a'\in\mathbb Q\\0\le a'<a}}
\left\{S:\min_{0\le x\le a'}H(S,x)=m(S)\right\}.
\]
右侧为可数个闭集的并，故 $x_*$ 可测。于是
$c_c^*(S)=c^{-}(S)+x_*(S)[c^{+}(S)-c^{-}(S)]$
是可测最小化选择，由定理\ref{thm:joint:representation}产生可行时间控制并达到下界。任意控制与逐点最小值之差非负，因此积分差为零当且仅当逐点差几乎处处为零。

固定 $S$，对式\eqref{eq:joint:cost-integrand}关于 $c$ 求偏导。将导数乘以正数
$c^3(cS-c_0\bar S)^2/S$，驻点方程化为
\begin{equation}
\begin{split}
0={}&w_c\frac{c^3}{c_0^2}(c-c_0)
\left(c+c_0-\frac{2c_0\bar S}{S}\right)\\
&+w_q\left(c-\frac{c_0S_c}{S}\right)
\left(-c^2+\frac{3c_0S_c}{S}c-
\frac{2c_0^2\bar S S_c}{S^2}\right).
\end{split}
\label{eq:joint:stationary}
\end{equation}
该多项式最高次系数为 $w_c/c_0^2>0$，不恒为零，至多有五个实根。连续可微函数在闭区间上的最小值只能在端点或内部驻点取得，故比较所列候选值即可；不需要假设 $f$ 单峰或凸。
\end{proof}

\subsection{端点的局部性质}
\label{sec:proof:endpoint-properties}
本小节证明命题\ref{prop:joint:global-structure}。
\begin{proof}
记 $n(c)=w_c(1-c/c_0)^2+w_q(1-c_0S_c/(cS))^2$。$f$ 关于 $c$ 的导数与
\[
n'(c)(cS-c_0\bar S)-S n(c)
\]
同号。在左端点 $c=c_0S_c/S$，该量为
\[
-2w_c\left(1-\frac{S_c}{S}\right)(S_c-\bar S)
-Sw_c\left(1-\frac{S_c}{S}\right)^2<0.
\]
故向允许区间内部增大 $c$ 即可降低成本。在右端点 $c=c_0$，该量为
\[
-\frac{w_q}{S}\left(1-\frac{S_c}{S}\right)
\bigl(S^2-3S_cS+2S_c\bar S\bigr).
\]
括号内二次式的较小根小于 $S_c$，较大根为 $S_{\rm sw}$，从而在 $S>S_{\rm sw}$ 时导数为负，在 $S_c<S<S_{\rm sw}$ 时导数为正。结合端点处的单侧比较即得 (i)(ii)。由定理\ref{thm:joint:cost-minimum}，成本最优控制几乎处处取逐状态最小点；它不取左端点，在 $S_c<S<\min\{S_{\rm sw},S^*\}$ 上也不取右端点，故得到命题最后的结论。
\end{proof}

\subsection{退出附近的控制组合}
\label{sec:proof:terminal-allocation}
本小节证明命题\ref{prop:joint:terminal-allocation}。
\begin{proof}
令 $z=1-S_c/S$，把允许接触率写为
$c=c_0(1-zx)$、$0\le x\le1$。于是
\[
\frac{q_c-q_0}{1-q_0}=\frac{z(1-x)}{1-zx},\qquad
\frac{f(S,c_0(1-zx))}{z^2}
=\frac{w_cx^2+w_q(1-x)^2/(1-zx)^2}
{c_0[S(1-zx)-\bar S]}.
\]
当 $S\downarrow S_c$ 时，右侧在 $x\in[0,1]$ 上一致收敛于
\[
\frac{w_cx^2+w_q(1-x)^2}{c_0(S_c-\bar S)},
\]
该函数严格凸，唯一最小化点为 $x=w_q/(w_c+w_q)$。由一致收敛和紧性，任何原问题最小化点序列均趋于 $w_q/(w_c+w_q)$：否则可取收敛子列，其极限将是极限函数的另一个最小化点，产生矛盾。因此所有最小化点在充分接近退出时均位于 $(0,1)$，并得到式\eqref{eq:joint:terminal-ratio}。

在这样一个正长度状态区间上，两种单控制分别对应 $x=0$ 和 $x=1$，均不能达到逐状态最小值；其非负成本差的积分因而严格为正。考虑满足 $c_{\min}<c_0$、$q_{\rm cap}>q_0$ 的能力约束时，充分接近 $S_c$ 处的允许区间仍为 $[c_0S_c/S,c_0]$，同一局部论证成立。
\end{proof}

\subsection{连续结构与权重单调性}
\label{sec:proof:cost-structure}
本小节证明定理\ref{thm:joint:cost-structure}、推论\ref{cor:joint:capacity-clip}与命题\ref{prop:joint:weight-monotone}。令 $\zeta=S/S_c$、$p=cS/(c_0S_c)$，则允许区间 $[c_0S_c/S,c_0]$ 对应 $p\in[1,\zeta]$：$p=1$ 为仅减少接触，$p=\zeta$ 为仅隔离控制。由 $c/c_0=p/\zeta$、$(q-q_0)/(1-q_0)=1-1/p$ 与 $cS-c_0\bar S=c_0S_c(p-b)$，
\[
f(S,c)=\frac{w_c}{c_0S_c}\,G(p;\zeta),\qquad
G(p;\zeta)=\frac{(1-p/\zeta)^2+r(1-1/p)^2}{p-b}.
\]
固定 $S$ 时 $p$ 是 $c$ 的严格增函数，只需讨论 $G(\cdot;\zeta)$。记 $\zeta_{\rm sw}=S_{\rm sw}/S_c=(3+\sqrt{9-8b})/2\in(2,3)$。直接计算得 $G_p=E/(p-b)^2$，其中
\[
E(p,\zeta)=\frac{p^2-2bp}{\zeta^2}+\frac{2b}{\zeta}-1+r\varphi(p),\qquad
\varphi(p)=\frac{(p-1)(3p-p^2-2b)}{p^3},
\]
并且
\begin{equation}
E_p=2(p-b)\Bigl[\frac1{\zeta^2}-r\psi(p)\Bigr],\quad
\psi(p)=\frac{2p-3}{p^4};\qquad
E_\zeta=-\frac{2[(p-b)^2+b(\zeta-b)]}{\zeta^3}<0.
\label{eq:proof:cost-E}
\end{equation}
两端的值为 $E(1,\zeta)=-(\zeta-1)(\zeta+1-2b)/\zeta^2<0$ 与 $E(\zeta,\zeta)=r\varphi(\zeta)$；后者在 $1<\zeta<\zeta_{\rm sw}$ 时为正，在 $\zeta>\zeta_{\rm sw}$ 时为负（命题\ref{prop:joint:global-structure}）。

\emph{内部驻点曲线。}由 $\varphi'(p)=-2(2p-3)(p-b)/p^4$，$\varphi$ 在 $(1,\zeta_{\rm sw})$ 上为正，最大值为 $\varphi(3/2)=(9-8b)/27$，在 $p\ge\zeta_{\rm sw}$ 时不为正。又 $(b+\sqrt{b^2-8b+8})^2\ge2(2-b)^2$，且 $8(9-8b)<27(2-b)^2$，故 $r_{\rm cont}<27/(9-8b)$。设 $r\le r_{\rm cont}$。固定 $p>1$，$E(p,\cdot)$ 在 $[p,\infty)$ 上严格减，由 $r\varphi(p)$ 降至 $r\varphi(p)-1<0$。因此满足 $1<p<\zeta$、$E(p,\zeta)=0$ 的点恰好组成一条曲线 $\zeta=\zeta_s(p)$，$p\in(1,\zeta_{\rm sw})$。由隐函数定理，$\zeta_s$ 连续可微，且 $\zeta_s'=-E_p/E_\zeta$，所以沿曲线 $\zeta_s'$ 与 $E_p$ 同号。$p\to1$ 或 $p\to\zeta_{\rm sw}$ 时 $E(p,p)=r\varphi(p)\to0$，而 $E_\zeta$ 在有界集上有负上界，故 $\zeta_s(p)-p\to0$，即 $\zeta_s(1+)=1$、$\zeta_s(\zeta_{\rm sw}-)=\zeta_{\rm sw}$。

\emph{曲线单调。}这一步证明 $\zeta_s$ 严格增：若曲线上某点 $E_p=0$，消元后得到 $F(p;\sqrt r)=0$；下面证明 $r\le r_{\rm cont}$ 时 $F$ 至多在 $r=r_{\rm cont}$、$p=2$ 处为零，从而曲线上 $E_p\ge0$。$p\le3/2$ 时 $\psi(p)\le0$，故 $E_p>0$。若曲线上某点 $p>3/2$ 处 $E_p=0$，则 $1/\zeta=\sqrt{r\psi(p)}$；代入 $E=0$ 并乘以 $-p^3$，得
\[
F(p;s):=s^2X(p)-2sbp\sqrt{2p-3}+p^3=0,\qquad X(p)=p^3-6p^2+6p+2b(3p-4),\qquad s=\sqrt r.
\]
下证：$0<s<s_c:=\sqrt{r_{\rm cont}}$ 时 $F>0$；$s=s_c$ 时 $F\ge0$，且等号仅在 $p=2$ 成立。由式\eqref{eq:joint:rcont}，$s_c=4/(b+\sqrt{b^2-8b+8})\in(\sqrt2,2)$ 满足 $(1-b)s_c^2+bs_c-2=0$，即 $b=(s_c^2-2)/[s_c(s_c-1)]$。令 $w=\sqrt{2p-3}$，代入并展开得
\[
F(p;s_c)=\frac{(w-1)^2\,Q(w,s_c)}{8(s_c-1)},
\]
\[
Q(w,s)=(s-1)(s^2+1)w^3(w+2)+12(s-1)w^2-2(s^3+3s^2-11s+3)w+(9s^2+11s-s^3-27).
\]
在 $s\in(\sqrt2,2)$ 上，常数项随 $s$ 增，不小于 $9(\sqrt2-1)$；一次项系数随 $s$ 减，不小于 $-2$；二次项系数大于 $12(\sqrt2-1)$；三、四次项非负。所以对 $w\ge0$ 有 $Q\ge9(\sqrt2-1)-2w+12(\sqrt2-1)w^2>0$（判别式为负），从而 $F(p;s_c)\ge0$，等号仅在 $w=1$ 即 $p=2$ 成立。对 $0<s<s_c$：$F$ 是 $s$ 的二次式且 $F(p;0)=p^3>0$。若 $X(p)\le0$，$F$ 在 $s>0$ 上严格减，故 $F(p;s)>F(p;s_c)\ge0$。若 $X(p)>0$ 且 $F$ 有实根 $s_1\le s_2$，则 $X\le b^2w^2/p$，于是
\[
s_2\ge\frac{bpw}{X}\ge\frac{p^2}{bw}>\frac{p^2}{\sqrt{2p-3}}\ge4>s_c;
\]
由 $F(p;s_c)\ge0$ 得 $s_c\le s_1$，故 $F(p;s)>0$。若 $F$ 无实根，亦有 $F>0$。因此 $r<r_{\rm cont}$ 时曲线上处处 $E_p\ne0$；$p$ 接近 $1$ 时 $E_p>0$，由连续性曲线上处处 $E_p>0$。令 $r\uparrow r_{\rm cont}$ 取极限，$r=r_{\rm cont}$ 时曲线上 $E_p\ge0$，零点只可能在 $p=2$。两种情形下 $\zeta_s$ 都在 $(1,\zeta_{\rm sw})$ 上严格增，是该区间到自身的双射。

\emph{定理\ref{thm:joint:cost-structure}的证明。}对 $1<\zeta<\zeta_{\rm sw}$，$E(\cdot,\zeta)$ 在 $(1,\zeta)$ 内恰有一个零点 $p^*(\zeta)=\zeta_s^{-1}(\zeta)$；又 $E(1,\zeta)<0<E(\zeta,\zeta)$，所以 $E$ 在 $p^*$ 处由负变正，$G(\cdot;\zeta)$ 先严格减、后严格增，最小点唯一且在内部。对 $\zeta\ge\zeta_{\rm sw}$，$E(\cdot,\zeta)$ 在 $(1,\zeta)$ 内没有零点且 $E(1,\zeta)<0$，所以 $G$ 严格减，最小点为 $p=\zeta$。$p^*$ 连续，且 $\zeta\uparrow\zeta_{\rm sw}$ 时 $p^*\to\zeta_{\rm sw}$，$\zeta\downarrow1$ 时 $p^*\to1$。回到原变量，$c_c^*(S)=c_0S_cp^*(S/S_c)/S$（$S<S_{\rm sw}$）与 $c_c^*(S)=c_0$（$S\ge S_{\rm sw}$）即定理所述；由定理\ref{thm:joint:cost-minimum}，成本最优控制几乎处处等于它。

\emph{推论\ref{cor:joint:capacity-clip}的证明。}能力约束下的允许区间 $[c^-(S),c^+(S)]\subset[c_0S_c/S,c_0]$ 对应 $p$ 的闭子区间。由上，$G(\cdot;\zeta)$ 在 $[1,\zeta]$ 上先严格减、后严格增（或严格减），它在任一闭子区间上的最小点唯一，等于 $p^*$ 截断到该子区间；换回 $c$ 即得式\eqref{eq:joint:capacity-clip}。

\emph{命题\ref{prop:joint:weight-monotone}的证明。}记 $G=A+rB$，$A=(1-p/\zeta)^2/(p-b)$，$B=(1-1/p)^2/(p-b)\ge0$。由于 $A_p=-(\zeta-p)(\zeta+p-2b)/[\zeta^2(p-b)^2]$，$A$ 在 $[1,\zeta]$ 上严格减。(i) 固定 $S$，设 $p_1$、$p_2$ 分别是 $r_1<r_2$ 下的最小点，二者属于同一允许区间。两个最优性不等式相加，得 $(r_2-r_1)[B(p_2)-B(p_1)]\le0$，即 $B(p_2)\le B(p_1)$。若 $p_2>p_1$，则 $A(p_2)<A(p_1)$，于是 $A(p_2)+r_1B(p_2)<A(p_1)+r_1B(p_1)$，与 $p_1$ 在 $r_1$ 下最优矛盾。故 $p_2\le p_1$，而 $c$、$q$ 都随 $p$ 增，即得逐状态结论。由式\eqref{eq:joint:duration}，接触率处处不增时控制时长不减；再由式\eqref{eq:joint:cumulative}与\eqref{eq:joint:quarantine}得到感染与隔离的结论。对任一允许控制，$J/w_c$ 是 $r$ 的仿射函数，斜率为非负的隔离强度积分，所以其下确界 $J_{\min}/w_c$ 是 $r$ 的凹且不减的函数。(ii) $G_{p\zeta}=E_\zeta/(p-b)^2<0$。设 $\zeta_1<\zeta_2$，$p_i$ 是 $\zeta_i$ 下的最小点，并假设 $p_2<p_1$。不考虑能力约束时允许区间为 $[1,\zeta]$，考虑时为 $[\max\{1,c_{\min}\zeta/c_0\},\min\{\zeta,(1-q_0)/(1-q_{\rm cap})\}]$，两端都随 $\zeta$ 不减，所以 $p_1$、$p_2$ 在两个状态上都允许。由最优性，
\[
[G(p_1;\zeta_2)-G(p_2;\zeta_2)]-[G(p_1;\zeta_1)-G(p_2;\zeta_1)]\ge0,
\]
而左侧等于 $\int_{\zeta_1}^{\zeta_2}\int_{p_2}^{p_1}G_{p\zeta}\,dp\,d\zeta<0$，矛盾。故 $p^*$ 随 $\zeta$ 不减，$q^*=1-(1-q_0)/p^*$ 亦然；阈值控制期内 $S$ 严格减，所以隔离比例随时间不增。

\subsection{给定时长的最小成本}
\label{sec:proof:duration-frontier}
本小节证明命题\ref{prop:joint:duration-frontier}。
\begin{proof}
由定理\ref{thm:joint:representation}，允许控制与取值于 $[c^-(S),c^+(S)]$ 的可测函数 $c_c$ 对应，且
\[
(\Delta t,J)[c_c]=\int_{S_c}^{S^*}g(S,c_c(S))\,dS,\qquad
g(S,c)=\frac{N}{\eta}\Bigl(\frac{1}{cS-c_0\bar S},\ f(S,c)\Bigr).
\]
$g$ 在 $\{(S,c):c^-(S)\le c\le c^+(S)\}$ 上连续且有界，因为分母不小于 $c_0(S_c-\bar S)>0$。令 $\Gamma(S)=\{g(S,c):c^-(S)\le c\le c^+(S)\}$，它取紧值且可积有界。由可测选择定理，$\Gamma$ 的每个可测选择都可写成 $g(S,c_c(S))$，其中 $c_c$ 可测。因此全部允许控制的 $(\Delta t,J)$ 组成的集合 $K$ 等于 $\Gamma$ 的积分。Lebesgue 测度无原子，由 \cite{Aumann1965} 的定理 1 与定理 4，$K$ 是紧凸集。

$K$ 在 $\Delta t$ 轴上的投影是 $[\Delta t_{\min},\Delta t_{\max}]$，所以对其中每个 $T$，$V(T)=\min\{J:(T,J)\in K\}$ 有定义且由紧性可以达到；$K$ 凸蕴含 $V$ 凸。对内点 $T$，凸函数 $V$ 有次梯度 $\nu$。令 $\kappa=-\nu$，则对任意 $(T',J')\in K$，
\[
J'+\kappa T'\ge V(T')+\kappa T'\ge V(T)+\kappa T,
\]
即达到 $V(T)$ 的控制使 $J+\kappa\Delta t$ 最小。把 $f$ 换成注记\ref{rem:joint:duration-constraint}中的被积函数后，重复定理\ref{thm:joint:cost-minimum}的论证，可知它几乎处处是逐状态最小化选择。
\end{proof}
