\section{仅隔离控制的补充结果}
\label{sec:sup:single-control}

\subsection{参数影响}
\label{sec:sup:contact-theory}
参数与偏导数的约定同正文第~\ref{sec:scaling-theory}~节。常规接触率首先决定控制是否被触发。
\begin{lemma}
\label{lem:s1:trigger-boundary}
固定其余参数，设 $I_0<\eta<I_0+\beta_2^0S_0/\beta_1^0$，则存在唯一的触发边界 $c_{0,\min}(\eta)$，满足
\begin{equation}
I_{\max}^{0}\bigl(c_{0,\min}(\eta)\bigr)=\eta,
\qquad c_{0,\min}(\eta)>\frac{\gamma N}{\beta(1-q_0)S_0},
\label{eq:s1:c0-min-def}
\end{equation}
且触发条件\eqref{eq:model:trigger}成立当且仅当 $c_0>c_{0,\min}(\eta)$。
\end{lemma}
\begin{proof}
当 $c_0\le\gamma N/[\beta(1-q_0)S_0]$，即 $S_0\le S_c$ 时，常规峰值为 $I_0$。$c_0$ 超过这一值后，$S_c$ 与 $\rho_1$ 都与 $c_0$ 成反比，$\beta_2^0/\beta_1^0$ 不含 $c_0$，由式\eqref{eq:s1:Imax-no}得
\begin{equation}
\frac{dI_{\max}^{0}}{dc_0}
=-\frac{\rho_1}{c_0}\ln\frac{S_c}{S_0}>0,
\label{eq:s1:Imax-no-c0-derivative}
\end{equation}
因此峰值随 $c_0$ 严格增加。$c_0\to\infty$ 时 $S_c\to0$、$\rho_1\ln(S_c/S_0)\to0$，故 $\lim_{c_0\to\infty}I_{\max}^{0}=I_0+\beta_2^0S_0/\beta_1^0$。$I_{\max}^{0}$ 关于 $c_0$ 连续，由 $I_0<\eta<I_0+\beta_2^0S_0/\beta_1^0$ 与严格单调性，$I_{\max}^{0}=\eta$ 恰有一个根，且该根大于 $\gamma N/[\beta(1-q_0)S_0]$。在 $I_0<\eta$ 下，触发条件\eqref{eq:model:trigger}等价于 $I_{\max}^{0}>\eta$，因而恰在 $c_0$ 大于该根时成立。
\end{proof}
\begin{proposition}
\label{prop:contact-effects}\label{cor:s1:duration-sensitivity}
设条件\eqref{eq:model:trigger}成立，则对仅隔离控制，
\begin{enumerate}
\renewcommand{\labelenumi}{(\roman{enumi})}
\item 启动时刻 $t_1$、启动时的隔离比例 $q_1$ 与控制时长 $\Delta t$ 满足
\[
\frac{\partial t_1}{\partial c_0}<0,\qquad
\frac{\partial q_1}{\partial c_0}>0,\qquad
\frac{\partial\Delta t}{\partial\beta}>0,\qquad
\frac{\partial\Delta t}{\partial q_0}<0;
\]
\item 控制时长对常规接触率的偏导数为
\begin{equation}
\frac{\partial\Delta t}{\partial c_0}
=\frac{N}{c_0^2\eta}
\left\{\frac{S^*}{S^*-\bar S}
\left[1+\frac{S_c}{S^*-S_c}\ln\frac{S_0}{S^*}\right]
-\ln\frac{S^*-\bar S}{S_c-\bar S}\right\},
\label{eq:duration-c0}
\end{equation}
它的符号不能统一为正或负；
\item 设 $I_0<\eta<I_0+\beta_2^0S_0/\beta_1^0$，固定除 $c_0$ 外的参数，则 $\Delta t$ 作为 $c_0$ 的函数在 $(c_{0,\min}(\eta),\infty)$ 内至少有一个极大值。
\end{enumerate}
\end{proposition}
\phantomsection
\label{sec:proof:contact-effects}
\begin{proof}
(i) 把式\eqref{eq:model:routine-orbit}中的 $I(S)$ 记为 $I(S;c_0)$，以显示它对 $c_0$ 的依赖。在 $S<S_0$ 时，
\[
\frac{\partial I(S;c_0)}{\partial c_0}=-\frac{\rho_1}{c_0}\ln\frac{S}{S_0}>0.
\]
在式\eqref{eq:s1:t1}中，积分前的系数 $1/\beta_1^0$ 随 $c_0$ 下降；由引理~\ref{lem:s1:Sstar-sensitivity}，下限 $S^*$ 随 $c_0$ 上升；由上式，被积函数 $1/[S\,I(S;c_0)]$ 也随 $c_0$ 下降。逐项求导均给出负项，故 $\partial t_1/\partial c_0<0$。对式\eqref{eq:s1:qmax}求导，得
\[
\frac{\partial q_1}{\partial c_0}
=\frac{\gamma N}{\beta c_0^2(S^*)^2}
\left(S^*+c_0\frac{\partial S^*}{\partial c_0}\right)>0.
\]
由式\eqref{eq:s1:t2}，$\Delta t=[N/(c_0\eta)]\ln[(S^*-\bar S)/(S_c-\bar S)]$，其中系数 $N/(c_0\eta)$ 不含 $\beta,q_0$。对 $\beta$ 求导时，
\[
\frac{\partial\bar S}{\partial\beta}=-\frac{\gamma N}{c_0\beta^2}<0,
\qquad
\frac{\partial(S_c-\bar S)}{\partial\beta}=-\frac{q_0S_c}{\beta}\le0,
\]
结合 $\partial S^*/\partial\beta>0$，对数的分子增大、分母不增，得 $\partial\Delta t/\partial\beta>0$。对 $q_0$ 求导时，$\bar S$ 不含 $q_0$，而 $\partial S^*/\partial q_0<0$、$\partial S_c/\partial q_0=S_c/(1-q_0)>0$，对数的分子减小、分母增大，得 $\partial\Delta t/\partial q_0<0$。

(ii) 利用 $\partial\bar S/\partial c_0=-\bar S/c_0$、$\partial S_c/\partial c_0=-S_c/c_0$ 及式\eqref{eq:s1:Sstar-eta-sign}中的 $\partial S^*/\partial c_0$，对式\eqref{eq:s1:t2}求导并整理，即得式\eqref{eq:duration-c0}。符号不定由 (iii) 的证明得到。

(iii) 当 $c_0$ 从右侧趋于 $c_{0,\min}(\eta)$ 时，常规峰值趋于 $\eta$，$S^*$ 趋于 $S_c$，故 $\Delta t$ 趋于零。$\beta_1^0/\beta_2^0$ 不含 $c_0$，由启动方程 $I(S^*)=\eta$，
$\lim_{c_0\to\infty}S^*=S_0-(\beta_1^0/\beta_2^0)(\eta-I_0)>0$，而 $S_c,\bar S$ 均为 $O(c_0^{-1})$，故
\[
\Delta t=\frac{N}{c_0\eta}[\ln c_0+O(1)],\qquad
\lim_{c_0\to\infty}\Delta t=0.
\]
$\Delta t$ 在 $(c_{0,\min}(\eta),\infty)$ 内为正且连续，两端极限为零，因此在内部取得正的最大值。由中值定理，其导数在该最大值点左侧至少一处为正、右侧至少一处为负，这也给出 (ii) 中的符号不定。
\end{proof}

\subsection{基准参数扫描}
\label{sec:sup:baseline-scan}
基准参数取自 Zhou 等\cite{Zhou2024SPCI}图 3 的示例。该例另设住院仓室，其非隔离感染者的自然康复率 $\gamma_I=0.1$ 与确诊率 $\delta_I=0.2504$ 之和对应本模型的移出率 $\gamma$；该例没有给出 $\delta_q$，取 $\delta_q=\gamma$。除 $c_0$ 与 $\eta/N$ 外，本小节参数取正文图~\ref{fig:scenario1:single-sim}的基准值。

常规接触率对控制时长的影响不是单调的（命题~\ref{prop:contact-effects}，图~\ref{fig:scenario1_summary_c0}）。$c_0$ 越大，常规轨迹越早触及阈值，$S^*$ 与 $q_1$ 越高。在所考察的范围内，控制时长随 $c_0$ 先增后减，基准阈值 $\eta/N=5\%$ 下（图中未画出）在 $c_0\approx5.1$ 处取得数值极大值约 $7.579$ d。当 $c_0$ 不超过触发边界 $c_{0,\min}(\eta)$（式\eqref{eq:s1:c0-min-def}）时，无需加强干预。该边界随阈值升高，基准阈值下约为 $3.327$。
